# ---- MyItineri web (React Native Web SPA) ----
# Upload as ".htaccess" in public_html, alongside index.html

<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # Force HTTPS
  RewriteCond %{HTTPS} off
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]

  # Serve real files/directories as-is (JS bundle, favicon, assets)
  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]

  # Everything else -> index.html so React Navigation can route it
  RewriteRule ^ index.html [L]
</IfModule>

# JS/CSS filenames are content-hashed by expo export, so they can be
# cached forever - a new build produces a new filename.
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType text/css              "access plus 1 year"
  ExpiresByType image/jpeg            "access plus 1 month"
  ExpiresByType image/png             "access plus 1 month"
</IfModule>

# index.html must NOT be cached, or visitors keep loading the old bundle
<IfModule mod_headers.c>
  <FilesMatch "index\.html$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json
</IfModule>
